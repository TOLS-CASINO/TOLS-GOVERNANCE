### Implementazione NextAuth

```typescript
// src/lib/auth.ts
import { NextAuthOptions } from "next-auth";
import CredentialsProvider from "next-auth/providers/credentials";
import { PrismaAdapter } from "@auth/prisma-adapter";
import { prisma } from "./db";
import bcrypt from "bcryptjs";
import { JWT } from "next-auth/jwt";

export const authOptions: NextAuthOptions = {
    adapter: PrismaAdapter(prisma),
    session: {
        strategy: "jwt",
        maxAge: 30 * 24 * 60 * 60, // 30 giorni
    },
    providers: [
        CredentialsProvider({
            name: "credentials",
            credentials: {
                email: { label: "Email", type: "email" },
                password: { label: "Password", type: "password" }
            },
            async authorize(credentials) {
                if (!credentials?.email || !credentials?.password) {
                    throw new Error("Credenziali mancanti");
                }

                const user = await prisma.user.findUnique({
                    where: { email: credentials.email }
                });

                if (!user || !user.passwordHash) {
                    throw new Error("Utente non trovato");
                }

                const isValid = await bcrypt.compare(
                    credentials.password,
                    user.passwordHash
                );

                if (!isValid) {
                    throw new Error("Password non valida");
                }

                return {
                    id: user.id,
                    email: user.email,
                    name: user.name,
                    role: user.role,
                    casinoId: user.casinoId
                };
            }
        })
    ],
    callbacks: {
        async jwt({ token, user }) {
            if (user) {
                token.id = user.id;
                token.role = user.role;
                token.casinoId = user.casinoId;
            }
            return token;
        },
        async session({ session, token }) {
            if (session.user) {
                session.user.id = token.id as string;
                session.user.role = token.role as string;
                session.user.casinoId = token.casinoId as string;
            }
            return session;
        }
    },
    pages: {
        signIn: '/login',
        error: '/login',
    }
};
```

### Middleware Protezione Route

```typescript
// src/middleware.ts
import { withAuth } from "next-auth/middleware";
import { NextResponse } from "next/server";

export default withAuth(
    function middleware(req) {
        const token = req.nextauth.token;
        const path = req.nextUrl.pathname;

        // Proteggi route admin
        if (path.startsWith('/admin') && token?.role !== 'admin') {
            return NextResponse.redirect(new URL('/dashboard', req.url));
        }

        // Proteggi route finance
        if (path.startsWith('/financial') && !['finance', 'controller', 'admin'].includes(token?.role as string)) {
            return NextResponse.redirect(new URL('/dashboard', req.url));
        }

        return NextResponse.next();
    },
    {
        callbacks: {
            authorized: ({ token }) => !!token
        }
    }
);

export const config = {
    matcher: [
        '/dashboard/:path*',
        '/financial/:path*',
        '/players/:path*',
        '/admin/:path*'
    ]
};
```

```typescript
// src/lib/tenant.ts
import { getServerSession } from "next-auth";
import { authOptions } from "./auth";
import { prisma } from "./db";

export async function getCurrentTenant() {
    const session = await getServerSession(authOptions);
    
    if (!session?.user?.casinoId) {
        throw new Error("Nessun tenant associato");
    }
    
    const casino = await prisma.casino.findUnique({
        where: { id: session.user.casinoId }
    });
    
    if (!casino || casino.status !== 'ACTIVE') {
        throw new Error("Tenant non attivo");
    }
    
    return casino;
}

export async function withTenantScope(
    table: string,
    query: any
) {
    const casino = await getCurrentTenant();
    
    return {
        ...query,
        where: {
            ...query.where,
            casinoId: casino.id
        }
    };
}
```

```typescript
"use client";

import React, { createContext, useContext, useEffect, useState } from 'react';
import { useSession } from 'next-auth/react';

interface TenantContextType {
    casino: Casino | null;
    loading: boolean;
    permissions: string[];
    switchCasino: (casinoId: string) => void;
}

const TenantContext = createContext<TenantContextType>({
    casino: null,
    loading: true,
    permissions: [],
    switchCasino: () => {}
});

export const TenantProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
    const { data: session } = useSession();
    const [casino, setCasino] = useState<Casino | null>(null);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        if (session?.user?.casinoId) {
            fetchCasino(session.user.casinoId);
        }
    }, [session]);

    const fetchCasino = async (casinoId: string) => {
        try {
            const response = await fetch(`/api/tenant/${casinoId}`);
            const data = await response.json();
            setCasino(data);
        } finally {
            setLoading(false);
        }
    };

    const permissions = casino?.permissions || [];

    return (
        <TenantContext.Provider value={{ casino, loading, permissions, switchCasino: fetchCasino }}>
            {children}
        </TenantContext.Provider>
    );
};

export const useTenant = () => useContext(TenantContext);
```

```typescript
export async function handleWebhook(req: Request) {
    const signature = req.headers.get('stripe-signature');
    const body = await req.text();

    if (!signature) throw new Error('Firma mancante');

    const event = stripe.webhooks.constructEvent(
        body,
        signature,
        process.env.STRIPE_WEBHOOK_SECRET!
    );

    switch (event.type) {
        case 'checkout.session.completed':
            await handleCheckoutCompleted(event.data.object);
            break;
        case 'invoice.paid':
            await handleInvoicePaid(event.data.object);
            break;
        case 'invoice.payment_failed':
            await handlePaymentFailed(event.data.object);
            break;
        case 'customer.subscription.deleted':
            await handleSubscriptionDeleted(event.data.object);
            break;
    }

    return { received: true };
}
```

### API Billing Routes

```typescript
// src/app/api/billing/route.ts
import { NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { createCheckoutSession, PLANS } from '@/lib/stripe';
import { prisma } from '@/lib/db';

export async function POST(req: Request) {
    const session = await getServerSession(authOptions);
    
    if (!session?.user?.casinoId) {
        return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 });
    }

    const { plan } = await req.json();

    try {
        const checkoutSession = await createCheckoutSession(
            session.user.casinoId,
            plan
        );

        return NextResponse.json({ url: checkoutSession.url });
    } catch (error) {
        return NextResponse.json(
            { error: 'Errore creazione sessione checkout' },
            { status: 500 }
        );
    }
}

export async function GET(req: Request) {
    const session = await getServerSession(authOptions);
    
    if (!session?.user?.casinoId) {
        return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 });
    }

    const subscription = await prisma.subscription.findUnique({
        where: { casinoId: session.user.casinoId },
        include: {
            invoices: true,
            payments: true
        }
    });

    return NextResponse.json({
        subscription,
        plans: PLANS
    });
}
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract EscrowAccount is Ownable {
    IERC20 public usdt;
    IERC20 public usdc;
    
    mapping(address => uint256) public balances;
    mapping(address => mapping(address => uint256)) public tokenBalances;
    
    event Deposited(address indexed from, address token, uint256 amount);
    event Withdrawn(address indexed to, address token, uint256 amount);
    event FundsReleased(address indexed to, address token, uint256 amount);
    
    constructor(address _usdt, address _usdc) {
        usdt = IERC20(_usdt);
        usdc = IERC20(_usdc);
    }
    
    function deposit(address token, uint256 amount) external {
        require(token == address(usdt) || token == address(usdc), "Token non supportato");
        require(amount > 0, "Importo non valido");
        
        require(IERC20(token).transferFrom(msg.sender, address(this), amount), "Transfer fallito");
        
        tokenBalances[token][msg.sender] += amount;
        emit Deposited(msg.sender, token, amount);
    }
    
    function releaseFunds(address to, address token, uint256 amount) external onlyOwner {
        require(tokenBalances[token][address(this)] >= amount, "Saldo insufficiente");
        
        tokenBalances[token][address(this)] -= amount;
        require(IERC20(token).transfer(to, amount), "Transfer fallito");
        
        emit FundsReleased(to, token, amount);
    }
}
```

### Web3 Service

```typescript
// src/services/web3.service.ts
import { ethers } from 'ethers';
import { EscrowAccountABI } from './abis/EscrowAccount';

export class Web3Service {
    private provider: ethers.providers.JsonRpcProvider;
    private signer: ethers.Signer;
    private escrowContract: ethers.Contract;

    constructor() {
        this.provider = new ethers.providers.JsonRpcProvider(
            process.env.NEXT_PUBLIC_RPC_URL
        );
        this.signer = this.provider.getSigner();
        this.escrowContract = new ethers.Contract(
            process.env.NEXT_PUBLIC_ESCROW_CONTRACT_ADDRESS!,
            EscrowAccountABI,
            this.signer
        );
    }

    async depositToEscrow(
        token: 'USDT' | 'USDC',
        amount: number
    ): Promise<ethers.providers.TransactionReceipt> {
        const tokenAddress = token === 'USDT' 
            ? process.env.NEXT_PUBLIC_USDT_ADDRESS!
            : process.env.NEXT_PUBLIC_USDC_ADDRESS!;
        
        const amountInWei = ethers.utils.parseUnits(amount.toString(), 6);
        
        // Approve spending
        const tokenContract = new ethers.Contract(
            tokenAddress,
            ['function approve(address spender, uint256 amount) returns (bool)'],
            this.signer
        );
        
        const approveTx = await tokenContract.approve(
            process.env.NEXT_PUBLIC_ESCROW_CONTRACT_ADDRESS!,
            amountInWei
        );
        await approveTx.wait();
        
        // Deposit
        const depositTx = await this.escrowContract.deposit(
            tokenAddress,
            amountInWei
        );
        
        return await depositTx.wait();
    }

    async getEscrowBalance(token: 'USDT' | 'USDC'): Promise<string> {
        const tokenAddress = token === 'USDT'
            ? process.env.NEXT_PUBLIC_USDT_ADDRESS!
            : process.env.NEXT_PUBLIC_USDC_ADDRESS!;
        
        const balance = await this.escrowContract.tokenBalances(
            tokenAddress,
            await this.signer.getAddress()
        );
        
        return ethers.utils.formatUnits(balance, 6);
    }
}
```

```python
@app.post("/predict/churn", response_model=ChurnPredictionResponse)
async def predict_churn(request: ChurnPredictionRequest):
    try:
        predictions = []
        
        for player in request.players:
            features = np.array([[
                player.session_frequency,
                player.deposit_amount,
                player.time_since_last_activity,
                player.bonus_utilization,
                player.support_interactions,
                player.average_bet,
                player.retention_days
            ]])
            
            probability = churn_model.predict_proba(features)[0][1]
            
            predictions.append({
                "player_id": player.player_id,
                "churn_probability": float(probability),
                "risk_level": get_risk_level(probability),
                "recommended_actions": get_retention_actions(probability)
            })
        
        return {"predictions": predictions}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/predict/ltv")
async def predict_ltv(player_data: PlayerData):

@app.post("/optimize/bonus")
async def optimize_bonus(data: Dict[str, Any]):
    try:
        budget = data.get("budget", 10000)
        players = data.get("players", [])
        
        if not players:
            return {"allocations": []}
        
        # Ottimizza allocazione bonus
        allocations = bonus_model.optimize(budget, players)
        
        return {
            "total_budget": budget,
            "allocations": allocations,
            "expected_roi": calculate_roi(allocations)
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/detect/fraud")
async def detect_fraud(transactions: List[Dict[str, Any]]):
    try:
        fraud_patterns = []
        
        for tx in transactions:
            features = extract_fraud_features(tx)
            score = fraud_model.predict_proba([features])[0][1]
            
            if score > 0.7:
                fraud_patterns.append({
                    "transaction_id": tx.get("id"),
                    "score": float(score),
                    "pattern": classify_fraud_pattern(features),
                    "recommended_action": get_fraud_action(score)
                })
        
        return {"fraud_patterns": fraud_patterns}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

def get_risk_level(probability: float) -> str:
    if probability > 0.7:
        return "high"
    elif probability > 0.4:
        return "medium"
    return "low"

def get_retention_actions(probability: float) -> List[str]:
    actions = []
    if probability > 0.7:
        actions = ["Send personalized bonus", "Contact VIP manager", "Schedule retention call"]
    elif probability > 0.4:
        actions = ["Send email campaign", "Offer free spins", "Push notification"]
    else:
        actions = ["Regular engagement", "Loyalty program"]
    return actions

def get_ltv_factors(player_data: PlayerData) -> List[Dict[str, Any]]:
    return [
        {"factor": "session_frequency", "impact": "high"},
        {"factor": "deposit_amount", "impact": "high"},
        {"factor": "average_bet", "impact": "medium"},
        {"factor": "retention_days", "impact": "medium"}
    ]

def calculate_roi(allocations: List[Dict]) -> float:
    total_investment = sum(a.get("amount", 0) for a in allocations)
    total_expected_return = sum(a.get("expected_return", 0) for a in allocations)
    return total_expected_return / total_investment if total_investment > 0 else 0

def extract_fraud_features(tx: Dict) -> List[float]:
    return [
        float(tx.get("amount", 0)),
        float(tx.get("frequency", 0)),
        float(tx.get("time_since_last", 0)),
        float(tx.get("location_score", 0)),
        float(tx.get("device_score", 0))
    ]

def classify_fraud_pattern(features: List[float]) -> str:
    if features[3] > 0.8:
        return "location_anomaly"
    elif features[4] > 0.8:
        return "device_anomaly"
    elif features[1] > 0.9:
        return "rapid_transactions"
    return "suspicious_activity"

def get_fraud_action(score: float) -> str:
    if score > 0.9:
        return "block_transaction"
    elif score > 0.7:
        return "flag_for_review"
    return "monitor"
```
