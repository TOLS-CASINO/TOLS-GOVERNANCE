    61→const CURRENCY_SYMBOLS: Record<string, string> = {
    62→  USD: '$', EUR: '€', GBP: '£', CAD: 'C$', AUD: 'A$',

    66→const fmtAmount = (amount: number, currency: string) => {
    67→  const sym = CURRENCY_SYMBOLS[currency] || currency
    68→  const isCrypto = ['BTC', 'ETH', 'USDC', 'USDT'].includes(currency)
    69→  if (isCrypto) return `${sym}${amount.toFixed(isCrypto && currency === 'BTC' ? 4 : 3)}`
    70→  return `${sym}${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    71→}
    72→
    73→const fmtUsd = (n: number) => `$${n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`

    75→function timeAgo(iso: string): string {
    76→  if (!iso) return '—'
    77→  const diff = Date.now() - new Date(iso).getTime()
    78→  const mins = Math.floor(diff / 60000)
    79→  if (mins < 1) return 'just now'
    80→  if (mins < 60) return `${mins} min ago`
    81→  const hrs = Math.floor(mins / 60)
    82→  if (hrs < 24) return `${hrs} hour${hrs > 1 ? 's' : ''} ago`
    83→  const days = Math.floor(hrs / 24)
    84→  return `${days} day${days > 1 ? 's' : ''} ago`
    85→}
