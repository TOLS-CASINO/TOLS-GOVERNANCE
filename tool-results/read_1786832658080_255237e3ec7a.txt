    30→/* ─── Currency helpers ─── */
    31→const CURRENCY_EMOJI: Record<string, string> = {
    32→  USD: '💵', EUR: '💶', GBP: '💷', BTC: '₿', ETH: '⟠', USDT: '₮', USDC: '◎', LTC: 'Ł',
    33→}
    34→
    35→const CRYPTO_CURRENCIES = ['BTC', 'ETH', 'LTC']
    36→
    37→function fmtCurrency(amount: number, currency: string): string {
    38→  if (CRYPTO_CURRENCIES.includes(currency)) {
    39→    return `${CURRENCY_EMOJI[currency] ?? ''} ${amount.toFixed(8)}`
    40→  }
    41→  const symbols: Record<string, string> = { USD: '$', EUR: '€', GBP: '£', USDT: '₮', USDC: '◎' }
    42→  const sym = symbols[currency] ?? currency
    43→  return `${sym}${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    44→}
    45→
    46→function fmtUsd(amount: number): string {
    47→  return `$${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    48→}
    49→
    50→function fmtDate(iso: string): string {
    51→  if (!iso) return '—'
    52→  const d = new Date(iso)
    53→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    54→}

   598→/* ═══════════════════════════════════════════════════
   599→   MAIN COMPONENT
   600→   ═══════════════════════════════════════════════════ */
   601→export function WalletsView() {
   602→  const { data, loading, error, refetch } = useApi(() => api.wallets.get())
   603→
   604→  /* All hooks at top level */
   605→  const [walletCurrencyFilter, setWalletCurrencyFilter] = useState('all')
   606→  const [walletStatusFilter, setWalletStatusFilter] = useState('all')
   607→  const [walletSearch, setWalletSearch] = useState('')
   608→  const [walletSort, setWalletSort] = useState<'balance-desc' | 'balance-asc'>('balance-desc')
   609→
   610→  const [txTypeFilter, setTxTypeFilter] = useState('all')
   611→  const [txCurrencyFilter, setTxCurrencyFilter] = useState('all')
   612→
   613→  const [toggledMethods, setToggledMethods] = useState<Set<string>>(new Set())
   614→
   615→  const toggleMethod = (id: string) => {
   616→    setToggledMethods(prev => {
   617→      const next = new Set(prev)
   618→      if (next.has(id)) next.delete(id)
   619→      else next.add(id)
   620→      return next
   621→    })
   622→  }
   623→
   624→  const wallets = data?.wallets ?? []
   625→  const transactions = data?.transactions ?? []
   626→  const currencyRates = data?.currencyRates ?? []
   627→  const totals = data?.totals
   628→  const paymentMethods = data?.paymentMethods ?? []
   629→
   630→  /* ── Filtered wallets ── */
   631→  const filteredWallets = useMemo(() => {
   632→    let result = [...wallets]
   633→    if (walletCurrencyFilter !== 'all') result = result.filter(w => w.currency === walletCurrencyFilter)
   634→    if (walletStatusFilter !== 'all') result = result.filter(w => w.status === walletStatusFilter)
   635→    if (walletSearch) {
   636→      const q = walletSearch.toLowerCase()
   637→      result = result.filter(w => w.playerName.toLowerCase().includes(q))
   638→    }
   639→    result.sort((a, b) => walletSort === 'balance-desc' ? b.balance - a.balance : a.balance - b.balance)
   640→    return result
   641→  }, [wallets, walletCurrencyFilter, walletStatusFilter, walletSearch, walletSort])
   642→
   643→  /* ── Filtered transactions ── */
   644→  const filteredTransactions = useMemo(() => {
   645→    let result = [...transactions]
   646→    if (txTypeFilter !== 'all') result = result.filter(t => t.type === txTypeFilter)
   647→    if (txCurrencyFilter !== 'all') result = result.filter(t => t.currency !== txCurrencyFilter)
   648→    return result
   649→  }, [transactions, txTypeFilter, txCurrencyFilter])
   650→
   651→  /* ── Unique values for filters ── */
   652→  const allCurrencies = useMemo(() => [...new Set(wallets.map(w => w.currency))].sort(), [wallets])
   653→  const allStatuses = useMemo(() => [...new Set(wallets.map(w => w.status))].sort(), [wallets])
   654→  const allTxTypes = useMemo(() => [...new Set(transactions.map(t => t.type))].sort(), [transactions])
   655→  const allTxCurrencies = useMemo(() => [...new Set(transactions.map(t => t.currency))].sort(), [transactions])
   656→
   657→  /* ── Wallet lookup ── */
   658→  const walletMap = useMemo(() => new Map(wallets.map(w => [w.id, w])), [wallets])
   659→
   660→  /* ── Currency breakdown ── */
   661→  const { pieData, barData, currencyEntries, totalBalanceAll } = useMemo(() => {
   662→    if (!totals) return { pieData: [] as { name: string; value: number; emoji: string }[], barData: [] as { currency: string; balance: number }[], currencyEntries: [] as [string, { balance: number; count: number }][], totalBalanceAll: 0 }
   663→    const entries = Object.entries(totals.byCurrency) as [string, { balance: number; count: number }][]
   664→    const totalBal = entries.reduce((sum, [, v]) => sum + v.balance, 0)
   665→    const pie = entries.map(([currency, info]) => ({
   666→      name: currency,
   667→      value: Number(info.balance.toFixed(2)),
   668→      emoji: CURRENCY_EMOJI[currency] ?? currency,
   669→    }))
   670→    const bar = entries.map(([currency, info]) => ({
   671→      currency: `${CURRENCY_EMOJI[currency] ?? ''} ${currency}`,
   672→      balance: Number(info.balance.toFixed(2)),
   673→    }))
   674→    return { pieData: pie, barData: bar, currencyEntries: entries, totalBalanceAll: totalBal }
   675→  }, [totals])
   676→
   677→  /* ── Loading ── */
   678→  if (loading) {
   679→    return (
   680→      <div className="space-y-4">
   681→        <Skeleton className="h-10 w-full max-w-md" />
   682→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   683→          {Array.from({ length: 4 }).map((_, i) => (
   684→            <Card key={i}><CardContent className="p-6"><Skeleton className="h-20 w-full" /></CardContent></Card>
   685→          ))}
   686→        </div>
   687→      </div>
   688→    )
   689→  }
   690→
   691→  /* ── Error ── */
   692→  if (error) {
   693→    return (
   694→      <Card className="border-destructive/40">
   695→        <CardContent className="p-6 text-center">
   696→          <Lock className="size-8 text-destructive mx-auto mb-2" />
   697→          <p className="text-sm font-medium text-destructive">Failed to load wallet data</p>
   698→          <p className="text-xs text-muted-foreground mt-1">{error}</p>
   699→          <Button variant="outline" size="sm" className="mt-3 gap-1" onClick={refetch}>
   700→            <RefreshCw className="size-3" /> Retry
   701→          </Button>
   702→        </CardContent>
   703→      </Card>
   704→    )
   705→  }
   706→
   707→  if (!data || !totals) return null
   708→
   709→  return (
   710→    <div className="space-y-4">
   711→      <div className="flex items-center justify-between">
   712→        <div className="flex items-center gap-2">
   713→          <Wallet className="size-5 text-amber-500" />
   714→          <h2 className="text-lg font-bold">Wallet Ecosystem</h2>
   715→        </div>
   716→        <Button variant="outline" size="sm" className="gap-1.5 text-xs" onClick={refetch}>
   717→          <RefreshCw className="size-3" /> Refresh
   718→        </Button>
   719→      </div>
   720→
   721→      <Tabs defaultValue="overview" className="w-full">
   722→        <TabsList className="h-8 bg-muted/50">
   723→          <TabsTrigger value="overview" className="text-xs data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-400">
   724→            Overview
   725→          </TabsTrigger>
   726→          <TabsTrigger value="wallets" className="text-xs data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-400">
   727→            All Wallets
   728→          </TabsTrigger>
   729→          <TabsTrigger value="transactions" className="text-xs data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-400">
   730→            Transactions
   731→          </TabsTrigger>
   732→          <TabsTrigger value="payments" className="text-xs data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-400">
   733→            Payment Methods
   734→          </TabsTrigger>
   735→        </TabsList>
   736→
   737→        <TabsContent value="overview" className="mt-4">
   738→          <OverviewTab
   739→            totals={totals}
   740→            currencyEntries={currencyEntries}
   741→            totalBalanceAll={totalBalanceAll}
   742→            pieData={pieData}
   743→            barData={barData}
   744→          />
   745→        </TabsContent>
   746→        <TabsContent value="wallets" className="mt-4">
   747→          <AllWalletsTab
   748→            filteredWallets={filteredWallets}
   749→            allCurrencies={allCurrencies}
   750→            allStatuses={allStatuses}
   751→            walletCurrencyFilter={walletCurrencyFilter}
   752→            setWalletCurrencyFilter={setWalletCurrencyFilter}
   753→            walletStatusFilter={walletStatusFilter}
   754→            setWalletStatusFilter={setWalletStatusFilter}
   755→            walletSearch={walletSearch}
   756→            setWalletSearch={setWalletSearch}
   757→            walletSort={walletSort}
   758→            setWalletSort={setWalletSort}
   759→          />
   760→        </TabsContent>
   761→        <TabsContent value="transactions" className="mt-4">
   762→          <TransactionsTab
   763→            filteredTransactions={filteredTransactions}
   764→            allTxTypes={allTxTypes}
   765→            allTxCurrencies={allTxCurrencies}
   766→            txTypeFilter={txTypeFilter}
   767→            setTxTypeFilter={setTxTypeFilter}
   768→            txCurrencyFilter={txCurrencyFilter}
   769→            setTxCurrencyFilter={setTxCurrencyFilter}
   770→            walletMap={walletMap}
   771→          />
   772→        </TabsContent>
   773→        <TabsContent value="payments" className="mt-4">
   774→          <PaymentMethodsTab
   775→            paymentMethods={paymentMethods}
   776→            currencyRates={currencyRates}
   777→            toggledMethods={toggledMethods}
   778→            toggleMethod={toggleMethod}
   779→          />
   780→        </TabsContent>
   781→      </Tabs>
   782→    </div>
   783→  )
