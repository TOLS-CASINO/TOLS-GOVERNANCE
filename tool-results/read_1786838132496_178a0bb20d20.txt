    93→function formatDuration(start: string, end: string): string {
    94→  const ms = new Date(end).getTime() - new Date(start).getTime()
    95→  const mins = Math.floor(ms / 60000)
    96→  if (mins < 60) return `${mins}m`
    97→  const hrs = Math.floor(mins / 60)
    98→  const remMins = mins % 60
    99→  return `${hrs}h ${remMins}m`
   100→}
   101→
   102→function formatTimeAgo(dateStr: string): string {
   103→  const diff = Date.now() - new Date(dateStr).getTime()
   104→  const secs = Math.floor(diff / 1000)
   105→  if (secs < 60) return `${secs}s ago`
   106→  const mins = Math.floor(secs / 60)
   107→  if (mins < 60) return `${mins}m ago`
   108→  const hrs = Math.floor(mins / 60)
   109→  return `${hrs}h ago`
   110→}
   111→
   112→function isNew(dateStr: string): boolean {
   113→  return Date.now() - new Date(dateStr).getTime() < 30000
   114→}
