   206→function relativeTime(dateStr: string | null | undefined): string {
   207→  if (!dateStr) return '—'
   208→  const diff = Date.now() - new Date(dateStr).getTime()
   209→  const mins = Math.floor(diff / 60000)
   210→  if (mins < 1) return 'just now'
   211→  if (mins < 60) return `${mins}m ago`
   212→  const hrs = Math.floor(mins / 60)
   213→  if (hrs < 24) return `${hrs}h ago`
   214→  const days = Math.floor(hrs / 24)
   215→  if (days < 30) return `${days}d ago`
   216→  const months = Math.floor(days / 30)
   217→  return `${months}mo ago`
   218→}
   219→
   220→function formatRevenue(n: number): string {
   221→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`
   222→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`
   223→  return `$${n.toFixed(0)}`
   224→}
   225→
   226→function maskValue(val: string | null | undefined): string {
   227→  if (!val) return '—'
   228→  if (val.length <= 8) return '••••••••'
   229→  return `${val.slice(0, 4)}${'•'.repeat(8)}${val.slice(-4)}`
   230→}
