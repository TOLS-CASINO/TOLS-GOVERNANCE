    30→/* ─── Currency helpers ─── */
    31→const CURRENCY_EMOJI: Record<string, string> = {
    32→  USD: '💵', EUR: '💶', GBP: '💷', BTC: '₿', ETH: '⟠', USDT: '₮', USDC: '◎', LTC: 'Ł',
    33→}
    34→
    35→const CRYPTO_CURRENCIES = ['BTC', 'ETH', 'LTC']
    36→
    37→function fmtCurrency(amount: number, currency: string): string {
    38→  if (CRYPTO_CURRENCIES.includes(currency)) {
    39→    return `${CURRENCY_EMOJI[currency] ?? ''} ${amount.toFixed(8)}`
    40→  }
    41→  const symbols: Record<string, string> = { USD: '$', EUR: '€', GBP: '£', USDT: '₮', USDC: '◎' }
    42→  const sym = symbols[currency] ?? currency
    43→  return `${sym}${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    44→}
    45→
    46→function fmtUsd(amount: number): string {
    47→  return `$${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    48→}
    49→
    50→function fmtDate(iso: string): string {
    51→  if (!iso) return '—'
    52→  const d = new Date(iso)
    53→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    54→}
