   358→function AddPaymentMethodDialog({ onAdd }: { onAdd: (method: PaymentMethod) => void }) {
   359→  const [type, setType] = useState<'card' | 'bank' | 'crypto'>('card')
   360→  const [label, setLabel] = useState('')
   361→  const [detail, setDetail] = useState('')
   362→  const [open, setOpen] = useState(false)
   363→
   364→  const handleAdd = () => {
   365→    if (!label || !detail) return
   366→    const newMethod: PaymentMethod = {
   367→      id: `pm-${Date.now()}`,
   368→      type,
   369→      label,
   370→      detail,
   371→      icon: CreditCard,
   372→      isDefault: false,
   373→    }
   374→    onAdd(newMethod)
   375→    setLabel('')
   376→    setDetail('')
   377→    setOpen(false)
   378→  }
   379→
   380→  return (
   381→    <Dialog open={open} onOpenChange={setOpen}>
   382→      <DialogTrigger asChild>
   383→        <Button size="sm" className="gap-1 text-xs">
   384→          <Plus className="size-3" /> Add Method
   385→        </Button>
   386→      </DialogTrigger>
   387→      <DialogContent className="sm:max-w-md">
   388→        <DialogHeader>
   389→          <DialogTitle className="text-primary">Add Payment Method</DialogTitle>
   390→        </DialogHeader>
   391→        <div className="space-y-4 pt-2">
   392→          <div className="space-y-2">
   393→            <Label className="text-xs text-muted-foreground">Type</Label>
   394→            <Select value={type} onValueChange={(v) => setType(v as 'card' | 'bank' | 'crypto')}>
   395→              <SelectTrigger className="text-xs">
   396→                <SelectValue />
   397→              </SelectTrigger>
   398→              <SelectContent>
   399→                <SelectItem value="card">Credit / Debit Card</SelectItem>
   400→                <SelectItem value="bank">Bank Account</SelectItem>
   401→                <SelectItem value="crypto">Crypto Wallet</SelectItem>
   402→              </SelectContent>
   403→            </Select>
   404→          </div>
   405→          <div className="space-y-2">
   406→            <Label className="text-xs text-muted-foreground">Label</Label>
   407→            <Input
   408→              placeholder={type === 'card' ? 'Visa' : type === 'bank' ? 'Chase' : 'MetaMask'}
   409→              value={label}
   410→              onChange={(e) => setLabel(e.target.value)}
   411→              className="text-xs"
   412→            />
   413→          </div>
   414→          <div className="space-y-2">
   415→            <Label className="text-xs text-muted-foreground">
   416→              {type === 'card' ? 'Card Number' : type === 'bank' ? 'Account Number' : 'Wallet Address'}
   417→            </Label>
   418→            <Input
   419→              placeholder={type === 'card' ? '4242 4242 4242 4242' : type === 'bank' ? '0000 0000' : '0x...'}
   420→              value={detail}
   421→              onChange={(e) => setDetail(e.target.value)}
   422→              className="text-xs"
   423→            />
   424→          </div>
   425→          <Button onClick={handleAdd} className="w-full gap-1 text-xs" disabled={!label || !detail}>
   426→            <Plus className="size-3" /> Add Payment Method
   427→          </Button>
   428→        </div>
   429→      </DialogContent>
