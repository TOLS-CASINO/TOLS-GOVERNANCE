    73→### Implementazione NextAuth
    74→
    75→```typescript
    76→// src/lib/auth.ts
    77→import { NextAuthOptions } from "next-auth";
    78→import CredentialsProvider from "next-auth/providers/credentials";
    79→import { PrismaAdapter } from "@auth/prisma-adapter";
    80→import { prisma } from "./db";
    81→import bcrypt from "bcryptjs";
    82→import { JWT } from "next-auth/jwt";
    83→
    84→export const authOptions: NextAuthOptions = {
    85→    adapter: PrismaAdapter(prisma),
    86→    session: {
    87→        strategy: "jwt",
    88→        maxAge: 30 * 24 * 60 * 60, // 30 giorni
    89→    },
    90→    providers: [
    91→        CredentialsProvider({
    92→            name: "credentials",
    93→            credentials: {
    94→                email: { label: "Email", type: "email" },
    95→                password: { label: "Password", type: "password" }
    96→            },
    97→            async authorize(credentials) {
    98→                if (!credentials?.email || !credentials?.password) {
    99→                    throw new Error("Credenziali mancanti");
   100→                }
   101→
   102→                const user = await prisma.user.findUnique({
   103→                    where: { email: credentials.email }
   104→                });
   105→
   106→                if (!user || !user.passwordHash) {
   107→                    throw new Error("Utente non trovato");
   108→                }
   109→
   110→                const isValid = await bcrypt.compare(
   111→                    credentials.password,
   112→                    user.passwordHash
   113→                );
   114→
   115→                if (!isValid) {
   116→                    throw new Error("Password non valida");
   117→                }
   118→
   119→                return {
   120→                    id: user.id,
   121→                    email: user.email,
   122→                    name: user.name,
   123→                    role: user.role,
   124→                    casinoId: user.casinoId
   125→                };
   126→            }
   127→        })
   128→    ],
   129→    callbacks: {
   130→        async jwt({ token, user }) {
   131→            if (user) {
   132→                token.id = user.id;
   133→                token.role = user.role;
   134→                token.casinoId = user.casinoId;
   135→            }
   136→            return token;
   137→        },
   138→        async session({ session, token }) {
   139→            if (session.user) {
   140→                session.user.id = token.id as string;
   141→                session.user.role = token.role as string;
   142→                session.user.casinoId = token.casinoId as string;
   143→            }
   144→            return session;
   145→        }
   146→    },
   147→    pages: {
   148→        signIn: '/login',
   149→        error: '/login',
   150→    }
   151→};
   152→```

   159→import { NextResponse } from "next/server";

   161→export default withAuth(
   162→    function middleware(req) {
   163→        const token = req.nextauth.token;
   164→        const path = req.nextUrl.pathname;
   165→
   166→        // Proteggi route admin
   167→        if (path.startsWith('/admin') && token?.role !== 'admin') {
   168→            return NextResponse.redirect(new URL('/dashboard', req.url));
   169→        }
   170→
   171→        // Proteggi route finance

   173→            return NextResponse.redirect(new URL('/dashboard', req.url));
   174→        }
   175→
   176→        return NextResponse.next();
   177→    },
   178→    {
   179→        callbacks: {
   180→            authorized: ({ token }) => !!token
   181→        }
   182→    }

   571→### API Billing Routes
   572→
   573→```typescript
   574→// src/app/api/billing/route.ts
   575→import { NextResponse } from 'next/server';
   576→import { getServerSession } from 'next-auth';
   577→import { authOptions } from '@/lib/auth';
   578→import { createCheckoutSession, PLANS } from '@/lib/stripe';
   579→import { prisma } from '@/lib/db';
   580→
   581→export async function POST(req: Request) {
   582→    const session = await getServerSession(authOptions);
   583→    
   584→    if (!session?.user?.casinoId) {
   585→        return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 });
   586→    }
   587→
   588→    const { plan } = await req.json();
   589→
   590→    try {
   591→        const checkoutSession = await createCheckoutSession(
   592→            session.user.casinoId,
   593→            plan
   594→        );
   595→
   596→        return NextResponse.json({ url: checkoutSession.url });
   597→    } catch (error) {
   598→        return NextResponse.json(
   599→            { error: 'Errore creazione sessione checkout' },
   600→            { status: 500 }
   601→        );
   602→    }
   603→}
   604→
   605→export async function GET(req: Request) {
   606→    const session = await getServerSession(authOptions);
   607→    
   608→    if (!session?.user?.casinoId) {
   609→        return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 });
   610→    }
   611→
   612→    const subscription = await prisma.subscription.findUnique({
   613→        where: { casinoId: session.user.casinoId },
   614→        include: {
   615→            invoices: true,
   616→            payments: true
   617→        }
   618→    });
   619→
   620→    return NextResponse.json({
   621→        subscription,
   622→        plans: PLANS
   623→    });
   624→}

   703→        setCurrentStep(prev => Math.max(prev - 1, 0));
   704→    };
   705→
   706→    const handleComplete = async () => {
   707→        try {
   708→            await fetch('/api/onboarding/complete', {
   709→                method: 'POST',
   710→                headers: { 'Content-Type': 'application/json' },
   711→                body: JSON.stringify(data)
   712→            });
   713→            router.push('/dashboard');
   714→        } catch (error) {
   715→            console.error('Errore completamento onboarding:', error);
   716→        }
   717→    };
   718→
   719→    const CurrentStep = steps[currentStep].component;
   720→
   721→    return (
   722→        <div className="min-h-screen bg-casino-bg flex items-center justify-center p-4">
   723→            <div className="w-full max-w-2xl">
   724→                {/* Progress Bar */}
   725→                <div className="mb-8">
   726→                    <div className="flex items-center justify-between mb-2">
   727→                        <h1 className="text-2xl font-bold text-white">
   728→                            {steps[currentStep].title}
   729→                        </h1>
   730→                        <span className="text-sm text-gray-400">
   731→                            Step {currentStep + 1} di {steps.length}
   732→                        </span>
   733→                    </div>
   734→                    <div className="w-full h-2 bg-casino-surface rounded-full">
   735→                        <div
   736→                            className="h-full bg-gradient-to-r from-purple-500 to-blue-500 rounded-full transition-all duration-500"
   737→                            style={{ width: `${((currentStep + 1) / steps.length) * 100}%` }}
   738→                        />
   739→                    </div>
   740→                </div>
   741→
   742→                {/* Step Content */}
   743→                <div className="bg-casino-surface border border-casino-border rounded-xl p-6 mb-6">
   744→                    <CurrentStep data={data} onNext={handleNext} />
   745→                </div>
   746→
   747→                {/* Navigation */}
   748→                <div className="flex justify-between">
   749→                    <button
   750→                        onClick={handleBack}
   751→                        disabled={currentStep === 0}
   752→                        className="px-6 py-3 bg-transparent border border-casino-border rounded-lg text-white disabled:opacity-50"
   753→                    >
   754→                        Indietro
   755→                    </button>
   756→                    {currentStep === steps.length - 1 ? (
   757→                        <button
   758→                            onClick={handleComplete}
   759→                            className="px-6 py-3 bg-gradient-to-r from-purple-600 to-blue-600 rounded-lg font-semibold"
   760→                        >
   761→                            Completa Setup
   762→                        </button>
   763→                    ) : (
   764→                        <button
   765→                            onClick={() => handleNext({})}
   766→                            className="px-6 py-3 bg-gradient-to-r from-purple-600 to-blue-600 rounded-lg font-semibold"
   767→                        >
   768→                            Avanti
   769→                        </button>
   770→                    )}
   771→                </div>
   772→            </div>
   773→        </div>
   774→    );
   775→};
   776→```

   785→// blockchain/contracts/EscrowAccount.sol
   786→// SPDX-License-Identifier: MIT
   787→pragma solidity ^0.8.19;
   788→
   789→import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
   790→import "@openzeppelin/contracts/access/Ownable.sol";
   791→
   792→contract EscrowAccount is Ownable {
   793→    IERC20 public usdt;
   794→    IERC20 public usdc;
   795→    
   796→    mapping(address => uint256) public balances;
   797→    mapping(address => mapping(address => uint256)) public tokenBalances;
   798→    
   799→    event Deposited(address indexed from, address token, uint256 amount);
   800→    event Withdrawn(address indexed to, address token, uint256 amount);
   801→    event FundsReleased(address indexed to, address token, uint256 amount);
   802→    
   803→    constructor(address _usdt, address _usdc) {
   804→        usdt = IERC20(_usdt);
   805→        usdc = IERC20(_usdc);
   806→    }
   807→    
   808→    function deposit(address token, uint256 amount) external {
   809→        require(token == address(usdt) || token == address(usdc), "Token non supportato");
   810→        require(amount > 0, "Importo non valido");
   811→        
   812→        require(IERC20(token).transferFrom(msg.sender, address(this), amount), "Transfer fallito");
   813→        
   814→        tokenBalances[token][msg.sender] += amount;
   815→        emit Deposited(msg.sender, token, amount);
   816→    }
   817→    
   818→    function releaseFunds(address to, address token, uint256 amount) external onlyOwner {
   819→        require(tokenBalances[token][address(this)] >= amount, "Saldo insufficiente");
   820→        
   821→        tokenBalances[token][address(this)] -= amount;
   822→        require(IERC20(token).transfer(to, amount), "Transfer fallito");
   823→        
   824→        emit FundsReleased(to, token, amount);
   825→    }
   826→}

   833→import { ethers } from 'ethers';

   836→export class Web3Service {

   853→    async depositToEscrow(
   854→        token: 'USDT' | 'USDC',
   855→        amount: number
   856→    ): Promise<ethers.providers.TransactionReceipt> {
   857→        const tokenAddress = token === 'USDT' 
   858→            ? process.env.NEXT_PUBLIC_USDT_ADDRESS!
   859→            : process.env.NEXT_PUBLIC_USDC_ADDRESS!;
   860→        
   861→        const amountInWei = ethers.utils.parseUnits(amount.toString(), 6);
   862→        
   863→        // Approve spending
   864→        const tokenContract = new ethers.Contract(
   865→            tokenAddress,
   866→            ['function approve(address spender, uint256 amount) returns (bool)'],
   867→            this.signer
   868→        );
   869→        
   870→        const approveTx = await tokenContract.approve(
   871→            process.env.NEXT_PUBLIC_ESCROW_CONTRACT_ADDRESS!,
   872→            amountInWei
   873→        );
   874→        await approveTx.wait();
   875→        
   876→        // Deposit
   877→        const depositTx = await this.escrowContract.deposit(
   878→            tokenAddress,
   879→            amountInWei
   880→        );
   881→        
   882→        return await depositTx.wait();
   883→    }
   884→
   885→    async getEscrowBalance(token: 'USDT' | 'USDC'): Promise<string> {
   886→        const tokenAddress = token === 'USDT'
   887→            ? process.env.NEXT_PUBLIC_USDT_ADDRESS!
   888→            : process.env.NEXT_PUBLIC_USDC_ADDRESS!;
   889→        
   890→        const balance = await this.escrowContract.tokenBalances(
   891→            tokenAddress,
   892→            await this.signer.getAddress()
   893→        );
   894→        
   895→        return ethers.utils.formatUnits(balance, 6);
   896→    }
   897→}

   904→### Python FastAPI Service
   905→
   906→```python
   907→# ai-engine/main.py
   908→from fastapi import FastAPI, HTTPException
   909→from pydantic import BaseModel
   910→from typing import List, Dict, Any
   911→import joblib
   912→import numpy as np
   913→from datetime import datetime, timedelta
   914→
   915→app = FastAPI(title="TOLS ML Engine")
   916→
   917→# Carica modelli
   918→churn_model = joblib.load("models/churn_model.pkl")
   919→ltv_model = joblib.load("models/ltv_model.pkl")
   920→bonus_model = joblib.load("models/bonus_optimizer.pkl")
   921→segment_model = joblib.load("models/segmenter.pkl")
   922→fraud_model = joblib.load("models/fraud_detector.pkl")
   923→
   924→class PlayerData(BaseModel):
   925→    player_id: str
   926→    session_frequency: float
   927→    deposit_amount: float
   928→    game_preference: str
   929→    time_since_last_activity: int
   930→    bonus_utilization: float
   931→    support_interactions: int
   932→    average_bet: float
   933→    retention_days: int
   934→
   935→class ChurnPredictionRequest(BaseModel):
   936→    players: List[PlayerData]
   937→
   938→class ChurnPredictionResponse(BaseModel):
   939→    predictions: List[Dict[str, Any]]
   940→
   941→@app.post("/predict/churn", response_model=ChurnPredictionResponse)
   942→async def predict_churn(request: ChurnPredictionRequest):
   943→    try:
   944→        predictions = []
   945→        
   946→        for player in request.players:
   947→            features = np.array([[
   948→                player.session_frequency,
   949→                player.deposit_amount,
   950→                player.time_since_last_activity,
   951→                player.bonus_utilization,
   952→                player.support_interactions,
   953→                player.average_bet,
   954→                player.retention_days
   955→            ]])
   956→            
   957→            probability = churn_model.predict_proba(features)[0][1]
   958→            
   959→            predictions.append({
   960→                "player_id": player.player_id,
   961→                "churn_probability": float(probability),
   962→                "risk_level": get_risk_level(probability),
   963→                "recommended_actions": get_retention_actions(probability)
   964→            })
   965→        
   966→        return {"predictions": predictions}
   967→    except Exception as e:
   968→        raise HTTPException(status_code=500, detail=str(e))
   969→
   970→@app.post("/predict/ltv")
   971→async def predict_ltv(player_data: PlayerData):
   972→    try:
   973→        features = np.array([[
   974→            player_data.session_frequency,
   975→            player_data.deposit_amount,
   976→            player_data.average_bet,
   977→            player_data.retention_days
   978→        ]])
   979→        
   980→        ltv = ltv_model.predict(features)[0]
   981→        
   982→        return {
   983→            "player_id": player_data.player_id,
   984→            "predicted_ltv": float(ltv),
   985→            "confidence": 0.85,
   986→            "factors": get_ltv_factors(player_data)
   987→        }
   988→    except Exception as e:
   989→        raise HTTPException(status_code=500, detail=str(e))
   990→
   991→@app.post("/optimize/bonus")
   992→async def optimize_bonus(data: Dict[str, Any]):
   993→    try:
   994→        budget = data.get("budget", 10000)
   995→        players = data.get("players", [])
   996→        
   997→        if not players:
   998→            return {"allocations": []}
   999→        
  1000→        # Ottimizza allocazione bonus
  1001→        allocations = bonus_model.optimize(budget, players)
  1002→        
  1003→        return {
  1004→            "total_budget": budget,
  1005→            "allocations": allocations,
  1006→            "expected_roi": calculate_roi(allocations)
  1007→        }
  1008→    except Exception as e:
  1009→        raise HTTPException(status_code=500, detail=str(e))
  1010→
  1011→@app.post("/detect/fraud")
  1012→async def detect_fraud(transactions: List[Dict[str, Any]]):
  1013→    try:
  1014→        fraud_patterns = []
  1015→        
  1016→        for tx in transactions:
  1017→            features = extract_fraud_features(tx)
  1018→            score = fraud_model.predict_proba([features])[0][1]
  1019→            
  1020→            if score > 0.7:
  1021→                fraud_patterns.append({
  1022→                    "transaction_id": tx.get("id"),
  1023→                    "score": float(score),
  1024→                    "pattern": classify_fraud_pattern(features),
  1025→                    "recommended_action": get_fraud_action(score)
  1026→                })
  1027→        
  1028→        return {"fraud_patterns": fraud_patterns}
  1029→    except Exception as e:
  1030→        raise HTTPException(status_code=500, detail=str(e))
  1031→
  1032→def get_risk_level(probability: float) -> str:
  1033→    if probability > 0.7:
  1034→        return "high"
  1035→    elif probability > 0.4:
  1036→        return "medium"
  1037→    return "low"
  1038→
  1039→def get_retention_actions(probability: float) -> List[str]:
  1040→    actions = []
  1041→    if probability > 0.7:
  1042→        actions = ["Send personalized bonus", "Contact VIP manager", "Schedule retention call"]
  1043→    elif probability > 0.4:
  1044→        actions = ["Send email campaign", "Offer free spins", "Push notification"]
  1045→    else:
  1046→        actions = ["Regular engagement", "Loyalty program"]
  1047→    return actions
  1048→
  1049→def get_ltv_factors(player_data: PlayerData) -> List[Dict[str, Any]]:
  1050→    return [
  1051→        {"factor": "session_frequency", "impact": "high"},
  1052→        {"factor": "deposit_amount", "impact": "high"},
  1053→        {"factor": "average_bet", "impact": "medium"},
  1054→        {"factor": "retention_days", "impact": "medium"}
  1055→    ]
  1056→
  1057→def calculate_roi(allocations: List[Dict]) -> float:
  1058→    total_investment = sum(a.get("amount", 0) for a in allocations)
  1059→    total_expected_return = sum(a.get("expected_return", 0) for a in allocations)
  1060→    return total_expected_return / total_investment if total_investment > 0 else 0
  1061→
  1062→def extract_fraud_features(tx: Dict) -> List[float]:
  1063→    return [
  1064→        float(tx.get("amount", 0)),
  1065→        float(tx.get("frequency", 0)),
  1066→        float(tx.get("time_since_last", 0)),
  1067→        float(tx.get("location_score", 0)),
  1068→        float(tx.get("device_score", 0))
  1069→    ]
  1070→
  1071→def classify_fraud_pattern(features: List[float]) -> str:
  1072→    if features[3] > 0.8:
  1073→        return "location_anomaly"
  1074→    elif features[4] > 0.8:
  1075→        return "device_anomaly"
  1076→    elif features[1] > 0.9:
  1077→        return "rapid_transactions"
  1078→    return "suspicious_activity"
  1079→
  1080→def get_fraud_action(score: float) -> str:
  1081→    if score > 0.9:
  1082→        return "block_transaction"
  1083→    elif score > 0.7:
  1084→        return "flag_for_review"
  1085→    return "monitor"
  1086→```
