   102→   102→/* ─── Animated Counter Hook ─── */
   103→   103→function useAnimatedCounter(target: number, duration = 1000): number {
   104→   104→  const [value, setValue] = useState(0)
   105→   105→  useEffect(() => {
   106→   106→    const start = value
   107→   107→    const diff = target - start
   108→   108→    if (diff === 0) return
   109→   109→    const startTime = performance.now()
   110→   110→    let raf: number
   111→   111→    const step = (now: number) => {
   112→   112→      const elapsed = now - startTime
   113→   113→      const progress = Math.min(elapsed / duration, 1)
   114→   114→      const eased = 1 - Math.pow(1 - progress, 3)
   115→   115→      setValue(Math.round(start + diff * eased))
   116→   116→      if (progress < 1) raf = requestAnimationFrame(step)
   117→   117→    }
   118→   118→    raf = requestAnimationFrame(step)
   119→   119→    return () => cancelAnimationFrame(raf)
   120→   120→  }, [target, duration])
   121→   121→  return value
   122→   122→}
