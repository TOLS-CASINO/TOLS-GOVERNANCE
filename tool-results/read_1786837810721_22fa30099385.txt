    91→function timeAgo(iso: string): string {
    92→  if (!iso) return '—'
    93→  const diff = Date.now() - new Date(iso).getTime()
    94→  const mins = Math.floor(diff / 60000)
    95→  if (mins < 1) return 'just now'
    96→  if (mins < 60) return `${mins}m ago`
    97→  const hrs = Math.floor(mins / 60)
    98→  if (hrs < 24) return `${hrs}h ago`
    99→  const days = Math.floor(hrs / 24)
   100→  return `${days}d ago`
   101→}
   102→
   103→function fmtDate(iso: string): string {
   104→  if (!iso) return '—'
   105→  return new Date(iso).toLocaleDateString('en-US', {
   106→    month: 'short', day: 'numeric', year: 'numeric',
   107→  })
   108→}
   109→
   110→function fmtNumber(n: number): string {
   111→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`
   112→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`
   113→  return n.toString()
   114→}

   362→function TokensTab({ tokens }: { tokens: ApiToken[] }) {
   363→  const [tokenList, setTokenList] = useState(tokens)

   365→  const [showRevokeDialog, setShowRevokeDialog] = useState<string | null>(null)
   366→  const [copiedId, setCopiedId] = useState<string | null>(null)
   367→  const [searchQuery, setSearchQuery] = useState('')
   368→  const [newTokenName, setNewTokenName] = useState('')
   369→  const [newTokenScopes, setNewTokenScopes] = useState<string[]>([])
   370→  const [newTokenRateLimit, setNewTokenRateLimit] = useState('1000')
   371→  const [generatedToken, setGeneratedToken] = useState<string | null>(null)
   372→
   373→  const scopeGroups = [
   374→    { module: 'Players', scopes: ['players:read', 'players:write'] },
   375→    { module: 'Wallets', scopes: ['wallets:read', 'wallets:write'] },
   376→    { module: 'Financial', scopes: ['financial:read', 'financial:write'] },
   377→    { module: 'Admin', scopes: ['admin'] },
   378→  ]
   379→
   380→  const allScopes = scopeGroups.flatMap(g => g.scopes)
   381→
   382→  const filteredTokens = useMemo(() => {
   383→    if (!searchQuery) return tokenList
   384→    const q = searchQuery.toLowerCase()
   385→    return tokenList.filter(t =>
   386→      t.name.toLowerCase().includes(q) ||
   387→      t.tokenPrefix.toLowerCase().includes(q) ||
   388→      t.scopes.some(s => s.toLowerCase().includes(q))
   389→    )
   390→  }, [tokenList, searchQuery])
   391→
   392→  function handleToggleActive(id: string) {

   395→
   396→  function handleCopyToken(id: string, prefix: string) {
   397→    navigator.clipboard.writeText(`${prefix}_masked_full_token`).catch(() => {})

   399→    setTimeout(() => setCopiedId(null), 2000)
   400→  }
   401→
   402→  function handleRevokeToken(id: string) {
   403→    setTokenList(prev => prev.filter(t => t.id !== id))
   404→    setShowRevokeDialog(null)
   405→  }
   406→
   407→  function handleGenerateToken() {
   408→    const token = generateFakeToken()
   409→    setGeneratedToken(token)
   410→  }
   411→
   412→  function handleCopyGenerated() {
   413→    if (generatedToken) {
   414→      navigator.clipboard.writeText(generatedToken).catch(() => {})
   415→    }
   416→  }
   417→
   418→  function handleAddToken() {
   419→    if (!newTokenName || newTokenScopes.length === 0) return
   420→    const prefix = `tols_${Math.random().toString(36).substring(2, 6)}`
   421→    const newToken: ApiToken = {
   422→      id: `tok_${Date.now()}`,
   423→      name: newTokenName,
   424→      tokenPrefix: prefix,
   425→      ownerId: 'owner_1',
   426→      scopes: newTokenScopes,
   427→      rateLimit: parseInt(newTokenRateLimit) || 1000,
   428→      isActive: true,
   429→      lastUsedAt: '',
   430→      expiresAt: new Date(Date.now() + 365 * 86400000).toISOString(),
   431→      createdAt: new Date().toISOString(),
   432→      usageCount: 0,
   433→      last24hCalls: 0,
   434→    }
   435→    setTokenList(prev => [newToken, ...prev])
   436→    resetNewDialog()
   437→  }
   438→
   439→  function resetNewDialog() {
   440→    setShowNewDialog(false)
   441→    setNewTokenName('')
   442→    setNewTokenScopes([])
   443→    setNewTokenRateLimit('1000')
   444→    setGeneratedToken(null)
   445→  }
   446→
   447→  function toggleScope(scope: string) {
   448→    setNewTokenScopes(prev =>
   449→      prev.includes(scope) ? prev.filter(s => s !== scope) : [...prev, scope]
   450→    )
   451→  }
   452→
   453→  return (
   454→    <div className="space-y-4">
   455→      {/* Header */}
   456→      <div className="flex items-center justify-between gap-2 flex-wrap">
   457→        <div className="flex items-center gap-2 flex-1 min-w-0">
   458→          <div className="relative flex-1 max-w-xs">
   459→            <Search className="absolute left-2.5 top-2 size-3.5 text-muted-foreground" />
   460→            <Input
   461→              placeholder="Search tokens..."
   462→              value={searchQuery}
   463→              onChange={e => setSearchQuery(e.target.value)}
   464→              className="pl-8 h-8 text-xs"
   465→            />
   466→          </div>
   467→          <Badge variant="secondary" className="text-[10px]">{tokenList.length} tokens</Badge>
   468→        </div>
   469→
   470→        <Dialog open={showNewDialog} onOpenChange={(open) => { if (!open) resetNewDialog(); setShowNewDialog(open) }}>
   471→          <DialogTrigger asChild>
   472→            <Button size="sm" className="gap-1.5">
   473→              <Plus className="size-3.5" />
   474→              Generate New Token
   475→            </Button>
   476→          </DialogTrigger>
   477→          <DialogContent className="sm:max-w-lg">
   478→            <DialogHeader>
   479→              <DialogTitle className="flex items-center gap-2">
   480→                <Key className="size-4 text-primary" />
   481→                Generate New API Token
   482→              </DialogTitle>
   483→              <DialogDescription>Create a new API token with specific scopes and rate limits.</DialogDescription>
   484→            </DialogHeader>
   485→
   486→            {!generatedToken ? (
