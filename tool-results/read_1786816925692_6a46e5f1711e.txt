   174→function relativeTime(dateStr: string): string {
   175→  const diff = Date.now() - new Date(dateStr).getTime()
   176→  const seconds = Math.floor(diff / 1000)
   177→  if (seconds < 60) return `${seconds}s ago`
   178→  const minutes = Math.floor(seconds / 60)
   179→  if (minutes < 60) return `${minutes}m ago`
   180→  const hours = Math.floor(minutes / 60)
   181→  if (hours < 24) return `${hours}h ago`
   182→  const days = Math.floor(hours / 24)
   183→  return `${days}d ago`
   184→}

   194→function highlightTemplateVars(text: string): React.ReactNode[] {
   195→  const parts = text.split(/(\{\{[^}]+\}\})/g)
   196→  return parts.map((part, i) => {
   197→    if (part.startsWith('{{') && part.endsWith('}}')) {
   198→      return (
   199→        <span key={i} className="bg-amber-500/20 text-amber-300 px-0.5 rounded font-mono text-xs">
   200→          {part}
   201→        </span>
   202→      )
   203→    }
   204→    return <span key={i}>{part}</span>
   205→  })
   206→}
