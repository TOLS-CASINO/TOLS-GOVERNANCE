   100→   100→
   101→   101→function isStepValid(step: number, data: FormData): boolean {
   102→   102→  switch (step) {
   103→   103→    case 0:
   104→   104→      return data.casinoName.trim().length >= 2
   105→   105→    case 1:
   106→   106→      return (
   107→   107→        data.casinoName.trim().length >= 2 &&
   108→   108→        data.primaryCurrency !== '' &&
   109→   109→        data.primaryLanguage !== '' &&
   110→   110→        data.timezone !== ''
   111→   111→      )
   112→   112→    case 2:
   113→   113→      return (
   114→   114→        data.adminFullName.trim().length >= 2 &&
   115→   115→        data.adminEmail.trim().length >= 5 &&
   116→   116→        data.adminEmail.includes('@') &&
   117→   117→        data.adminPassword.length >= 6 &&
   118→   118→        data.adminPassword === data.adminConfirmPassword
   119→   119→      )
   120→   120→    case 3:
   121→   121→      return true // integrations are optional
   122→   122→    case 4:
   123→   123→      return true
   124→   124→    default:
   125→   125→      return false
   126→   126→  }
   127→   127→}
