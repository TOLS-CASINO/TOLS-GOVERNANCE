    99→// ─── Step Validation ─────────────────────────────────────────────────────────
   100→
   101→function isStepValid(step: number, data: FormData): boolean {
   102→  switch (step) {
   103→    case 0:
   104→      return data.casinoName.trim().length >= 2
   105→    case 1:
   106→      return (
   107→        data.casinoName.trim().length >= 2 &&
   108→        data.primaryCurrency !== '' &&
   109→        data.primaryLanguage !== '' &&
   110→        data.timezone !== ''
   111→      )
   112→    case 2:
   113→      return (
   114→        data.adminFullName.trim().length >= 2 &&
   115→        data.adminEmail.trim().length >= 5 &&
   116→        data.adminEmail.includes('@') &&
   117→        data.adminPassword.length >= 6 &&
   118→        data.adminPassword === data.adminConfirmPassword
   119→      )
   120→    case 3:
   121→      return true // integrations are optional
   122→    case 4:
   123→      return true
   124→    default:
   125→      return false
   126→  }
   127→}
