   109→function formatDuration(start: string, end: string): string {
   110→  const ms = new Date(end).getTime() - new Date(start).getTime()
   111→  const mins = Math.floor(ms / 60000)
   112→  if (mins < 60) return `${mins}m`
   113→  const hrs = Math.floor(mins / 60)
   114→  const remMins = mins % 60
   115→  return `${hrs}h ${remMins}m`
   116→}
   117→
   118→function formatTimeAgo(dateStr: string): string {
   119→  const diff = Date.now() - new Date(dateStr).getTime()
   120→  const secs = Math.floor(diff / 1000)
   121→  if (secs < 60) return `${secs}s ago`
   122→  const mins = Math.floor(secs / 60)
   123→  if (mins < 60) return `${mins}m ago`
   124→  const hrs = Math.floor(mins / 60)
   125→  return `${hrs}h ago`
   126→}
   127→
   128→function isNew(dateStr: string): boolean {
   129→  return Date.now() - new Date(dateStr).getTime() < 30000
   130→}
