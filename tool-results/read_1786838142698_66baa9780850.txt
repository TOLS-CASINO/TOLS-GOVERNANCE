    30→    30→/* ─── Currency helpers ─── */
    31→    31→const CURRENCY_EMOJI: Record<string, string> = {
    32→    32→  USD: '💵', EUR: '💶', GBP: '💷', BTC: '₿', ETH: '⟠', USDT: '₮', USDC: '◎', LTC: 'Ł',
    33→    33→}
    34→    34→
    35→    35→const CRYPTO_CURRENCIES = ['BTC', 'ETH', 'LTC']
    36→    36→
    37→    37→function fmtCurrency(amount: number, currency: string): string {
    38→    38→  if (CRYPTO_CURRENCIES.includes(currency)) {
    39→    39→    return `${CURRENCY_EMOJI[currency] ?? ''} ${amount.toFixed(8)}`
    40→    40→  }
    41→    41→  const symbols: Record<string, string> = { USD: '$', EUR: '€', GBP: '£', USDT: '₮', USDC: '◎' }
    42→    42→  const sym = symbols[currency] ?? currency
    43→    43→  return `${sym}${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    44→    44→}
    45→    45→
    46→    46→function fmtUsd(amount: number): string {
    47→    47→  return `$${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    48→    48→}
    49→    49→
    50→    50→function fmtDate(iso: string): string {
    51→    51→  if (!iso) return '—'
    52→    52→  const d = new Date(iso)
    53→    53→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    54→    54→}

   601→   601→export function WalletsView() {
   602→   602→  const { data, loading, error, refetch } = useApi(() => api.wallets.get())
   603→   603→
   604→   604→  /* All hooks at top level */
   605→   605→  const [walletCurrencyFilter, setWalletCurrencyFilter] = useState('all')
   606→   606→  const [walletStatusFilter, setWalletStatusFilter] = useState('all')
   607→   607→  const [walletSearch, setWalletSearch] = useState('')
   608→   608→  const [walletSort, setWalletSort] = useState<'balance-desc' | 'balance-asc'>('balance-desc')
   609→   609→
   610→   610→  const [txTypeFilter, setTxTypeFilter] = useState('all')
   611→   611→  const [txCurrencyFilter, setTxCurrencyFilter] = useState('all')
   612→   612→
   613→   613→  const [toggledMethods, setToggledMethods] = useState<Set<string>>(new Set())
   614→   614→
   615→   615→  const toggleMethod = (id: string) => {
   616→   616→    setToggledMethods(prev => {
   617→   617→      const next = new Set(prev)
   618→   618→      if (next.has(id)) next.delete(id)
   619→   619→      else next.add(id)
   620→   620→      return next
   621→   621→    })
   622→   622→  }
   623→   623→
   624→   624→  const wallets = data?.wallets ?? []
   625→   625→  const transactions = data?.transactions ?? []
   626→   626→  const currencyRates = data?.currencyRates ?? []
   627→   627→  const totals = data?.totals

   631→   631→  const filteredWallets = useMemo(() => {
   632→   632→    let result = [...wallets]
   633→   633→    if (walletCurrencyFilter !== 'all') result = result.filter(w => w.currency === walletCurrencyFilter)
   634→   634→    if (walletStatusFilter !== 'all') result = result.filter(w => w.status === walletStatusFilter)
   635→   635→    if (walletSearch) {
   636→   636→      const q = walletSearch.toLowerCase()
   637→   637→      result = result.filter(w => w.playerName.toLowerCase().includes(q))
   638→   638→    }
   639→   639→    result.sort((a, b) => walletSort === 'balance-desc' ? b.balance - a.balance : a.balance - b.balance)
   640→   640→    return result
   641→   641→  }, [wallets, walletCurrencyFilter, walletStatusFilter, walletSearch, walletSort])
   642→   642→
   643→   643→  /* ── Filtered transactions ── */
   644→   644→  const filteredTransactions = useMemo(() => {
   645→   645→    let result = [...transactions]
   646→   646→    if (txTypeFilter !== 'all') result = result.filter(t => t.type === txTypeFilter)
   647→   647→    if (txCurrencyFilter !== 'all') result = result.filter(t => t.currency !== txCurrencyFilter)
   648→   648→    return result
   649→   649→  }, [transactions, txTypeFilter, txCurrencyFilter])
   650→   650→
   651→   651→  /* ── Unique values for filters ── */
   652→   652→  const allCurrencies = useMemo(() => [...new Set(wallets.map(w => w.currency))].sort(), [wallets])
   653→   653→  const allStatuses = useMemo(() => [...new Set(wallets.map(w => w.status))].sort(), [wallets])
   654→   654→  const allTxTypes = useMemo(() => [...new Set(transactions.map(t => t.type))].sort(), [transactions])
   655→   655→  const allTxCurrencies = useMemo(() => [...new Set(transactions.map(t => t.currency))].sort(), [transactions])
   656→   656→
   657→   657→  /* ── Wallet lookup ── */
   658→   658→  const walletMap = useMemo(() => new Map(wallets.map(w => [w.id, w])), [wallets])
   659→   659→
   660→   660→  /* ── Currency breakdown ── */
   661→   661→  const { pieData, barData, currencyEntries, totalBalanceAll } = useMemo(() => {

   663→   663→    const entries = Object.entries(totals.byCurrency) as [string, { balance: number; count: number }][]
   664→   664→    const totalBal = entries.reduce((sum, [, v]) => sum + v.balance, 0)

   666→   666→      name: currency,
   667→   667→      value: Number(info.balance.toFixed(2)),
   668→   668→      emoji: CURRENCY_EMOJI[currency] ?? currency,
   669→   669→    }))
   670→   670→    const bar = entries.map(([currency, info]) => ({

   672→   672→      balance: Number(info.balance.toFixed(2)),
   673→   673→    }))
   674→   674→    return { pieData: pie, barData: bar, currencyEntries: entries, totalBalanceAll: totalBal }
   675→   675→  }, [totals])
   676→   676→
   677→   677→  /* ── Loading ── */
   678→   678→  if (loading) {
   679→   679→    return (
   680→   680→      <div className="space-y-4">
   681→   681→        <Skeleton className="h-10 w-full max-w-md" />
   682→   682→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">

   689→   689→  }

   691→   691→  /* ── Error ── */

   693→   693→    return (
   694→   694→      <Card className="border-destructive/40">
   695→   695→        <CardContent className="p-6 text-center">

   699→   699→          <Button variant="outline" size="sm" className="mt-3 gap-1" onClick={refetch}>
   700→   700→            <RefreshCw className="size-3" /> Retry

   702→   702→        </CardContent>
   703→   703→      </Card>
   704→   704→    )
   705→   705→  }
   706→   706→
   707→   707→  if (!data || !totals) return null
   708→   708→
   709→   709→  return (
   710→   710→    <div className="space-y-4">
   711→   711→      <div className="flex items-center justify-between">
   712→   712→        <div className="flex items-center gap-2">
   713→   713→          <Wallet className="size-5 text-amber-500" />
   714→   714→          <h2 className="text-lg font-bold">Wallet Ecosystem</h2>
   715→   715→        </div>
   716→   716→        <Button variant="outline" size="sm" className="gap-1.5 text-xs" onClick={refetch}>
   717→   717→          <RefreshCw className="size-3" /> Refresh
   718→   718→        </Button>
   719→   719→      </div>
   720→   720→
   721→   721→      <Tabs defaultValue="overview" className="w-full">
