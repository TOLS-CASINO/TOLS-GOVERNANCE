   441→   441→function truncateAddress(addr: string, chars: number = 8): string {
   442→   442→  if (addr.length <= chars * 2 + 3) return addr
   443→   443→  return `${addr.slice(0, chars + 2)}...${addr.slice(-chars)}`
   444→   444→}

   446→   446→function formatUsd(value: number): string {
   447→   447→  return new Intl.NumberFormat('en-US', {
   448→   448→    style: 'currency',
   449→   449→    currency: 'USD',
   450→   450→    minimumFractionDigits: 0,
   451→   451→    maximumFractionDigits: 0,
   452→   452→  }).format(value)
   453→   453→}

   455→   455→function formatBalance(value: number): string {
   456→   456→  if (value >= 1000) return new Intl.NumberFormat('en-US').format(value)
   457→   457→  return value.toString()
