   313→
   314→  const [modelFilter, setModelFilter] = useState<string>('all')
   315→  const [confidenceThreshold, setConfidenceThreshold] = useState<string>('0')
   316→  const [addModelOpen, setAddModelOpen] = useState(false)
   317→  const [newModel, setNewModel] = useState({ name: '', type: 'classification' as ModelType, algorithm: '', description: '' })
   318→  const [searchQuery, setSearchQuery] = useState('')
   319→
   320→  /* useApi for data fetching */
   321→  const { data: pipelineData, loading, error, refetch } = useApi(
   322→    useCallback(() => api.dashboard.get().then(() => 'loaded'), []),
   323→    []
   324→  )
   325→
   326→  /* Simulate progress for running training jobs */
   327→  const [runningProgress, setRunningProgress] = useState<Record<string, number>>({
   328→    'job-2847': 67,
   329→    'job-2846': 34,
   330→  })
   331→
   332→  useEffect(() => {
   333→    const interval = setInterval(() => {
   334→      setRunningProgress((prev) => {
   335→        const next = { ...prev }
   336→        for (const key of Object.keys(next)) {
   337→          if (next[key] < 100) {
   338→            next[key] = Math.min(100, next[key] + Math.random() * 2)
   339→          }
   340→        }
   341→        return next
   342→      })
   343→    }, 1500)
   344→    return () => clearInterval(interval)
   345→  }, [])
   346→
   347→  /* Filtered predictions */
   348→  const filteredPredictions = predictions.filter((p) => {
   349→    if (modelFilter !== 'all' && p.model !== modelFilter) return false
   350→    if (Number(confidenceThreshold) > 0 && p.confidence < Number(confidenceThreshold)) return false
   351→    return true
   352→  })
   353→
   354→  /* Handle model action */
   355→  const handleModelAction = (modelId: string, action: 'deploy' | 'train' | 'view') => {
   356→    if (action === 'deploy') {
   357→      setModels((prev) =>
   358→        prev.map((m) => (m.id === modelId ? { ...m, status: 'deployed' as ModelStatus } : m))
   359→      )
   360→    } else if (action === 'train') {
   361→      setModels((prev) =>
   362→        prev.map((m) =>
   363→          m.id === modelId ? { ...m, status: 'training' as ModelStatus, trainingProgress: 0 } : m
   364→        )
   365→      )
   366→    }
   367→  }
   368→
   369→  /* Handle feature store toggle */
   370→  const handleFeatureToggle = (storeId: string, active: boolean) => {
   371→    setFeatureStores((prev) =>
   372→      prev.map((fs) => (fs.id === storeId ? { ...fs, active } : fs))
   373→    )
   374→  }
   375→
   376→  /* Handle cancel job */
   377→  const handleCancelJob = (jobId: string) => {
   378→    setTrainingJobs((prev) =>
   379→      prev.map((j) => (j.id === jobId ? { ...j, status: 'failed' as JobStatus } : j))
   380→    )
   381→    setRunningProgress((prev) => {
   382→      const next = { ...prev }
   383→      delete next[jobId]
   384→      return next
   385→    })
   386→  }
   387→
   388→  /* Loading state */
   389→  if (loading) {
   390→    return (
   391→      <div className="space-y-4">
   392→        <div className="flex items-center gap-2">
   393→          <Brain className="size-5 text-amber-400" />
   394→          <Skeleton className="h-6 w-48" />
   395→        </div>
   396→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   397→          {Array.from({ length: 6 }).map((_, i) => (
   398→            <Card key={i} className="border-zinc-800 bg-zinc-950/80">
   399→              <CardContent className="p-6">
   400→                <Skeleton className="h-32 w-full" />
   401→              </CardContent>
   402→            </Card>
   403→          ))}
   404→        </div>
   405→      </div>
   406→    )
   407→  }
   408→
   409→  return (
   410→    <div className="space-y-4">
   411→      {/* Header */}
   412→      <div className="flex items-center justify-between">
   413→        <div className="flex items-center gap-3">
   414→          <div className="flex items-center justify-center size-10 rounded-xl bg-gradient-to-br from-amber-500/20 to-rose-500/20 border border-amber-500/20">
   415→            <Brain className="size-5 text-amber-400" />
   416→          </div>
   417→          <div>
   418→            <h2 className="text-lg font-semibold text-zinc-100">ML Pipeline & Predictions</h2>
   419→            <p className="text-xs text-zinc-500">
   420→              {models.filter((m) => m.status === 'deployed').length} deployed ·{' '}
   421→              {models.filter((m) => m.status === 'training').length} training ·{' '}
   422→              {predictions.length} predictions/min
   423→            </p>
   424→          </div>
   425→        </div>
   426→        <Dialog open={addModelOpen} onOpenChange={setAddModelOpen}>
   427→          <DialogTrigger asChild>
   428→            <Button
   429→              size="sm"
   430→              className="gap-1.5 text-xs bg-gradient-to-r from-amber-600 to-amber-500 hover:from-amber-500 hover:to-amber-400 text-zinc-950 font-semibold border-0"
   431→            >
   432→              <Plus className="size-3.5" /> Add Model
   433→            </Button>
   434→          </DialogTrigger>
   435→          <DialogContent className="bg-zinc-950 border-zinc-800">
   436→            <DialogHeader>
   437→              <DialogTitle className="text-zinc-100">Add New ML Model</DialogTitle>

   440→              <div>
   441→                <Label className="text-xs text-zinc-400 mb-1.5 block">Model Name</Label>
   442→                <Input
   443→                  value={newModel.name}
   444→                  onChange={(e) => setNewModel((p) => ({ ...p, name: e.target.value }))}
   445→                  placeholder="e.g. Bonus Abuse Detector"
   446→                  className="h-9 text-xs bg-zinc-900 border-zinc-800 text-zinc-100 placeholder:text-zinc-600"
   447→                />
   448→              </div>
   449→              <div>
