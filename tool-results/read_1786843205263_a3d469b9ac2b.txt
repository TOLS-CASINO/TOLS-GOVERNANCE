    69→function truncateAddr(addr: string, start = 8, end = 6): string {
    70→  if (!addr || addr.length <= start + end + 3) return addr
    71→  return `${addr.slice(0, start)}...${addr.slice(-end)}`
    72→}

    74→function formatTimeAgo(dateStr: string) {
    75→  const diff = Date.now() - new Date(dateStr).getTime()
    76→  const days = Math.floor(diff / 86400000)
    77→  if (days > 30) return `${Math.floor(days / 30)}mo ago`
    78→  if (days > 0) return `${days}d ago`
    79→  const hours = Math.floor(diff / 3600000)
    80→  if (hours > 0) return `${hours}h ago`
    81→  const mins = Math.floor(diff / 60000)
    82→  if (mins > 0) return `${mins}m ago`
    83→  return 'just now'
    84→}
    85→
    86→function formatShortDate(dateStr: string) {
    87→  const d = new Date(dateStr)
    88→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    89→}

   106→function mapVipLevel(vip: string): VipLevel {
   107→  const lvl = parseInt(vip, 10)
   108→  if (lvl >= 1 && lvl <= 5) return lvl as VipLevel
   109→  const lower = vip.toLowerCase()
   110→  if (lower.includes('elite') || lower.includes('platinum')) return 5
   111→  if (lower.includes('gold') || lower.includes('vip')) return 4
   112→  if (lower.includes('silver')) return 3
   113→  if (lower.includes('bronze')) return 2
   114→  return 1
   115→}
