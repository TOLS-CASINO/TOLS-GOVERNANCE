    91→    91→function timeAgo(iso: string): string {
    92→    92→  if (!iso) return '—'
    93→    93→  const diff = Date.now() - new Date(iso).getTime()
    94→    94→  const mins = Math.floor(diff / 60000)
    95→    95→  if (mins < 1) return 'just now'
    96→    96→  if (mins < 60) return `${mins}m ago`
    97→    97→  const hrs = Math.floor(mins / 60)
    98→    98→  if (hrs < 24) return `${hrs}h ago`
    99→    99→  const days = Math.floor(hrs / 24)
   100→   100→  return `${days}d ago`
   101→   101→}
   102→   102→
   103→   103→function fmtDate(iso: string): string {
   104→   104→  if (!iso) return '—'
   105→   105→  return new Date(iso).toLocaleDateString('en-US', {
   106→   106→    month: 'short', day: 'numeric', year: 'numeric',
   107→   107→  })
   108→   108→}
   109→   109→
   110→   110→function fmtNumber(n: number): string {
   111→   111→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`
   112→   112→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`
   113→   113→  return n.toString()
   114→   114→}
